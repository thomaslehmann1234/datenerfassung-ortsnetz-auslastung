+*00:05:00 {
  # Ortsnetz-Auslastung fuer FHEM, Version fhem-0.2
  # Einfuegen im DEF-Feld des at-Devices (einfache Semikolons, siehe README)

  # ---- Konfiguration ---------------------------------------------------
  my $lat   = 51.123;      # Breitengrad, ungefaehr (3 Nachkommastellen)
  my $lon   = 10.123;      # Laengengrad, ungefaehr (3 Nachkommastellen)

  # Pro Phase [Device, Reading mit der Spannung in Volt].
  # Nicht gemessene Phase: [] eintragen, sie wird als -1 gesendet.
  my @src = (
    ['Shelly_L1', 'voltage'],
    ['Shelly_L2', 'voltage'],
    ['Shelly_L3', 'voltage'],
  );

  my @freq  = ('Shelly_L1', 'frequency');   # Netzfrequenz in Hz, () wenn nicht vorhanden
  my $alle  = 1;    # 1 = nur senden, wenn alle konfigurierten Phasen gueltig sind
  my $model = '';   # optional, z. B. 'Shelly Plus 1PM' (wird einmalig gesendet)
  # ---------------------------------------------------------------------

  my $name = $SELF // 'at_ortsnetz';
  my (@u, $fehlt);
  my $i = 0;
  foreach my $s (@src) {
    $i++;
    if (!@$s) { push @u, -1; next; }
    my $v = ReadingsNum($s->[0], $s->[1], 0);
    readingsSingleUpdate($defs{$name}, "l${i}_v", $v, 0);
    if ($v > 0 && $v <= 500 && ReadingsAge($s->[0], $s->[1], 99999) <= 600) {
      push @u, $v + 0;
    } else {
      push @u, -1;
      $fehlt = 1;
    }
  }
  if (($alle && $fehlt) || !grep { $_ > 0 } @u) {
    readingsSingleUpdate($defs{$name}, 'api_status', 'nicht gesendet: Spannung ungueltig oder aelter als 10 min', 1);
    return;
  }

  my %p = (
    observed_at => POSIX::strftime('%Y-%m-%dT%H:%M:%SZ', gmtime()),
    latitude    => $lat + 0,
    longitude   => $lon + 0,
    l1_v => $u[0], l2_v => $u[1], l3_v => $u[2],
    integration_version => 'fhem-0.2',
  );
  if (@freq) {
    my $f = ReadingsNum($freq[0], $freq[1], 0);
    $p{grid_frequency_hz} = $f + 0 if ($f >= 45 && $f <= 55 && ReadingsAge($freq[0], $freq[1], 99999) <= 600);
  }
  $p{smartmeter_model} = $model if ($model && ReadingsVal($name, 'model_gesendet', '') ne $model);

  require JSON::PP;
  my $json = JSON::PP->new->canonical->encode(\%p);

  HttpUtils_NonblockingGet({
    url      => 'https://www.ortsnetz-auslastung.de/v1/measurements',
    method   => 'POST',
    header   => 'Content-Type: application/json',
    data     => $json,
    timeout  => 15,
    callback => sub {
      my ($h, $err, $data) = @_;
      my $hash = $defs{$name};
      return if (!$hash);
      readingsBeginUpdate($hash);
      if ($err || ($h->{code} // 0) != 202) {
        my $res = $err ? "Fehler $err" : "HTTP " . ($h->{code} // '?');
        readingsBulkUpdate($hash, 'api_status', $res);
        Log3 $name, 3, "$name: $res " . substr($data // '', 0, 300);
      } else {
        my $r  = eval { JSON::PP::decode_json($data) } || {};
        my $st = $r->{status} || {};
        my $rd = $r->{redispatch} || {};
        readingsBulkUpdate($hash, 'api_status', 'HTTP 202');
        readingsBulkUpdate($hash, 'ampel', $st->{overall} // '?');
        readingsBulkUpdate($hash, "ampel_l$_", $st->{"l$_"} // '?') for (1 .. 3);
        readingsBulkUpdate($hash, 'speicher_empfehlung', $r->{storage_recommendation} // '?');
        readingsBulkUpdate($hash, 'redispatch', defined $rd->{active} ? ($rd->{active} ? 'ja' : 'nein') : 'unbekannt');
        readingsBulkUpdate($hash, 'model_gesendet', $p{smartmeter_model}) if ($p{smartmeter_model});
        Log3 $name, 5, "$name: gesendet $json, Antwort $data";
      }
      readingsEndUpdate($hash, 1);
    },
  });
  return;
}
